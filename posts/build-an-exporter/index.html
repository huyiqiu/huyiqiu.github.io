编写一个简单的prometheus exporter

exporter 是什么

exporter可以被简单地认为是一个提供监控指标的工具，对于prometheus生态，exporter是一个http服务器，prometheus通过轮询的方式向这个服务器抓取数据。

开源社区已经有了各种各样的exporter，用来监控各种常见的系统，但是总有一些指标是不能覆盖的，因此有时候我们需要自己来实现一个exporter，用来监控我们想要的性能指标。

如何实现一个exporter

prometheus提供了基于各种语言的客户端，使得我们能够更加方便的实现一个exporter，接下来我们以go为例，实现一个简单的exporter，用来监控GPU上的进程

定义指标

首先我们需要定义一个指标，prometheus接受的指标有一个标准的规范，一条指标通常包括三个内容，分别是帮助信息、类型信息、指标内容

我们的目标是定义一个监控GPU上的进程信息，所有我们希望指标的样子应该是这样

type Exporter struct {
	mutex           sync.RWMutex
	procMetricsDesc *prometheus.Desc
}

func NewExporter() *Exporter {
	return &Exporter{
		procMetricsDesc: prometheus.NewDesc("running_process_mem_used_bytes",
			"running processes memory used",
			[]string{"uuid", "pid"}, nil),
	}
}

type Metric struct {
	Uuid  string
	Pid   string
	Value float64
}

func (e *Exporter) Describe(descCh chan<- *prometheus.Desc) {
	descCh <- e.procMetricsDesc
}

func (e *Exporter) Collect(metricCh chan<- prometheus.Metric) {
	e.mutex.Lock()
	defer e.mutex.Unlock()
	metrics := scrape()
	for _, m := range metrics {
		metricInfo, _ := prometheus.NewConstMetric(e.procMetricsDesc, prometheus.GaugeValue, m.Value, m.Uuid, m.Pid)
		metricCh <- metricInfo
	}
}

func scrape() []Metric {
	result := make([]Metric, 0)
  // 获取进程信息
	procs := nvidia.Fetch() 
	for _, p := range procs {
		metric := Metric{
			Uuid: p.Uuid,
			Pid:  fmt.Sprintf("%d", p.Pid),
			Value: float64(p.MemUsed),
		}
		result = append(result, metric)
	}
	return result
}

package nvidia

import "github.com/NVIDIA/go-nvml/pkg/nvml"

type ProcInfo struct {
	Uuid    string
	Pid     uint32
	MemUsed uint64
}

func Fetch() []ProcInfo {
	nvml.Init()
	defer nvml.Shutdown()
	result := make([]ProcInfo, 0)
	count, _ := nvml.DeviceGetCount()
	for i := 0; i < count; i ++ {
		device, _ := nvml.DeviceGetHandleByIndex(i)
		uuid, _ := device.GetUUID()
		computeProcs, _ := device.GetComputeRunningProcesses()
		for _, item := range computeProcs {
			proc := ProcInfo {
				Uuid: uuid,
				Pid: item.Pid,
				MemUsed: item.UsedGpuMemory,
			}
			result = append(result, proc)
		}
		graphicPorcs, _ := device.GetGraphicsRunningProcesses()
		for _, item := range graphicPorcs {
			proc := ProcInfo {
				Uuid: uuid,
				Pid: item.Pid,
				MemUsed: item.UsedGpuMemory,
			}
			result = append(result, proc)
		}
	}
	return result
}
